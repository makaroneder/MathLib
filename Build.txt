MakeObjects(dir : String, ext : String, cmd : String, sources : String) : String = DeclareTarget("$(BUILDDIR)/" + dir + "/%." + ext + ".o", "$(SRCDIR)/" + dir + "/%." + ext + " $(HEADERS)", cmd, false) + CreateObjects("$(BUILDDIR)/" + dir + "/%." + ext + ".o", "$(SRCDIR)/" + dir + "/%." + ext, sources)
LDTarget(out : String, objs : String, libs : String, ld : String, ldFlags : String, deps : String) : String = DeclareTarget(out, objs + deps, ld + " " + ldFlags + " " + objs + " -o $@ " + libs, false)
ARTarget(out : String, objs : String, ar : String) : String  = DeclareTarget(out, objs, ar + " -rcs $@ $^", false)
CPTarget(out : String, in : String) : String = DeclareTarget(out, in, "cp $< $@", false)

DefaultRunTargets(name : String, app : String, args : String) : Void = DeclareTarget("run" + name, app, "./$< " + args, true) + DeclareTarget("debug" + name, app, "$(VALGRIND) $(VALGRINDFLAGS) ./$< " + args, true)
DefaultTarget(out : String, dir : String, libs : String, run : String, args : String, runDeps : String, buildDeps : String) : Void = {
    LDTarget("$(BUILDDIR)/" + out, MakeObjects(dir, "cpp", "$(CXX) $(CXXFLAGS) -c $< -o $@", FindFiles("$(SRCDIR)/" + dir, ".cpp")), "-L $(BUILDDIR) -l Math " + libs, "$(CXX)", "$(CXXFLAGS)", "$(BUILDDIR)/libMath.so " + buildDeps),
    DefaultRunTargets(run, "$(BUILDDIR)/" + out + " " + runDeps, args)
}

kernelObjs : String =
    "$(BUILDDIR)/OS/Shared/crti.s.o $(shell $(OSCXX) $(OSCXXFLAGS) -print-file-name=crtbegin.o) " +
    MakeObjects("OS/Kernel", "cpp", "$(OSCXX) $(OSCXXFLAGS) -c $< -o $@", FindFiles("$(SRCDIR)/OS/Kernel", ".cpp")) +
    MakeObjects("OS/Kernel", "asm", "$(AS) $(ASFLAGS) $< -o $@", FindFiles("$(SRCDIR)/OS/Kernel", ".asm")) +

    "$(shell $(OSCXX) $(OSCXXFLAGS) -print-file-name=crtend.o) $(BUILDDIR)/OS/Shared/crtn.s.o "
LDTarget("$(OSSYSROOT)/boot/Kernel.elf", kernelObjs, "-l gcc", "$(OSCXX)", "$(OSLDFLAGS) -T $(OSLINKER) -Xlinker -Map=$(BUILDDIR)/Kernel.map", "$(OSLINKER)")
